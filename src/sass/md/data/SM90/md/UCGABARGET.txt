FORMAT PREDICATE @[!]UniformPredicate(UPT):UPg Opcode
UniformRegister:URd
$( { '&' REQ:req '=' BITSET(6/0x0000):req_bit_set } )$
$( { '?' USCHED_INFO("DRAIN"):usched_info } )$
$( { '?' BATCH_T("NOP"):batch_t } )$
$( { '?' PM_PRED("PMN"):pm_pred } )$;

CONDITIONS

ILLEGAL_INSTR_ENCODING_ERROR
 (%SHADER_TYPE == $ST_UNKNOWN) || ((%SHADER_TYPE == $ST_TRAP)||(%SHADER_TYPE == $ST_CS)) :
 "Can only be used with CS"

 (((URd)==`UniformRegister@URZ)||((URd)<=(%MAX_UNIFORM_REG_COUNT-1))) :
 "Uniform register URd is out of range"
ILLEGAL_INSTR_ENCODING_SASS_ONLY_ERROR
 DEFINED TABLES_opex_1(batch_t,usched_info) :
 "Invalid combination of batch_t, usched_info"

 MEM_SCBD = NONE ;

 IDEST_SIZE = 32;

        UCGABARGETudp_pipe =  0b1010111000111;
        UCGABARGET =  0b1010111000111;

!ucgabarget__unused;
BITS_3_14_12_Pg = UPg;

BITS_13_91_91_11_0_opcode=Opcode;

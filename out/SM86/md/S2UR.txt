CLASS "s2ur_"
FORMAT PREDICATE @[!]UniformPredicate(UPT):UPg Opcode
UniformRegister:URd
','SpecialRegister:SRa
$( { '&' REQ:req '=' BITSET(6/0x0000):req_bit_set } )$
$( { '&' RD:rd '=' UImm(3/0x7):src_rel_sb } )$
$( { '&' WR:wr '=' UImm(3/0x7):dst_wr_sb } )$
$( { '?' USCHED_INFO("DRAIN"):usched_info } )$
$( { '?' BATCH_T("NOP"):batch_t } )$
$( { '?' PM_PRED("PMN"):pm_pred } )$;

CONDITIONS

ILLEGAL_INSTR_ENCODING_SASS_ONLY_ERROR
 ((%SHADER_TYPE != $ST_TRAP) && (%SHADER_TYPE != $ST_UNKNOWN)) -> ((SRa!=84) && (SRa!=85)) :
 "Special registers 84 and 85 should not be read in user mode"
OOR_REG_ERROR
 (((URd)==`UniformRegister@URZ)||((URd)<=(%MAX_UNIFORM_REG_COUNT-1))) :
 "Uniform register URd is out of range"
ILLEGAL_INSTR_ENCODING_SASS_ONLY_ERROR
 DEFINED TABLES_opex_0(batch_t,usched_info) :
 "Invalid combination of batch_t, usched_info"


PROPERTIES
 INSTRUCTION_TYPE = INST_TYPE_DECOUPLED_RD_WR_SCBD;
 MIN_WAIT_NEEDED = 1 ;
 VALID_IN_SHADERS = ISHADER_ALL ;

PREDICATES
 IDEST_SIZE = 32;
 ISRC_A_SIZE = 32;
 VIRTUAL_QUEUE = $VQ_SR2UR ;


OPCODES
        S2URudp_pipe =  0b100111000011;
        S2UR =  0b100111000011;

ENCODING
!s2ur__unused;
BITS_3_14_12_Pg = UPg;
BITS_1_15_15_Pg_not = UPg@not;
BITS_13_91_91_11_0_opcode=Opcode;
BITS_6_21_16_URd=URd;
BITS_8_79_72_imm8=SRa;
BITS_6_121_116_req_bit_set=req_bit_set;
BITS_3_115_113_src_rel_sb=VarLatOperandEnc(src_rel_sb);
BITS_3_112_110_dst_wr_sb=VarLatOperandEnc(dst_wr_sb);
BITS_2_103_102_pm_pred=pm_pred;
BITS_8_124_122_109_105_opex=TABLES_opex_0(batch_t,usched_info);
